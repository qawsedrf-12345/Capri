local function rastrear(c) conexoes[#conexoes + 1] = c; return c end
local function rf(a, b) return a + math.random() * (b - a) end
local function pick(t) return t[math.random(#t)] end

local function seguro(fn, ...)
    local args = table.pack(...)
    task.spawn(function()
        local ok, err = pcall(fn, table.unpack(args, 1, args.n))
        if not ok then
            warn("[Capri] erro:", err)
            local erros = genv.__CapriErros or {}
            genv.__CapriErros = erros
            erros[#erros + 1] = tostring(err)
        end
    end)
end

local function novo(classe, props, pai)
    local o = Instance.new(classe)
    for k, v in pairs(props) do o[k] = v end
    if pai then o.Parent = pai end
    return o
end

local function tw(obj, dur, props, estilo, dir)
    local t = TweenService:Create(obj, TweenInfo.new(dur, estilo or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local CAM_BIND = "CapriCamFX"
local CamFX = { shakes = {}, fov = nil, congelar = nil, custom = nil, ligado = false }

local function camPasso()
    local c = workspace.CurrentCamera
    if not c then return end
    local t = os.clock()

    if CamFX.congelar then
        c.CFrame = CamFX.congelar
        c.Focus = CFrame.new(CamFX.congelar.Position)
    end

    if CamFX.fov then
        local f = CamFX.fov
        local a = (t - f.ini) / f.dur
        if a >= 1 then
            c.FieldOfView = f.para
            CamFX.fov = nil
        else
            c.FieldOfView = f.de + (f.para - f.de) * TweenService:GetValue(a, f.estilo, f.dir)
        end
    end

    if CamFX.custom then
        local okc = pcall(CamFX.custom, c, t)
        if not okc then CamFX.custom = nil end
    end

    local ox, oy, rot = 0, 0, 0
    for i = #CamFX.shakes, 1, -1 do
        local s = CamFX.shakes[i]
        local p = (t - s.ini) / s.dur
        if p >= 1 then
            table.remove(CamFX.shakes, i)
        else
            local f = (1 - p) * (1 - p) * s.intens
            ox = ox + (math.random() - 0.5) * 2 * f
            oy = oy + (math.random() - 0.5) * 2 * f
            if s.rot then rot = rot + (math.random() - 0.5) * math.rad(2.2) * f end
        end
    end
    if ox ~= 0 or oy ~= 0 or rot ~= 0 then
        c.CFrame = c.CFrame * CFrame.new(ox, oy, 0) * CFrame.Angles(0, 0, rot)
    end

    if not CamFX.congelar and not CamFX.fov and not CamFX.custom and #CamFX.shakes == 0 then
        CamFX.ligado = false
        pcall(function() RunService:UnbindFromRenderStep(CAM_BIND) end)
    end
end

local function camLigar()
    if CamFX.ligado then return end
    CamFX.ligado = true
    RunService:BindToRenderStep(CAM_BIND, Enum.RenderPriority.Camera.Value + 2, camPasso)
end

local function camShake(intens, dur, rot)
    CamFX.shakes[#CamFX.shakes + 1] = { ini = os.clock(), dur = dur, intens = intens, rot = rot }
    camLigar()
end

local function camFov(de, para, dur, estilo, dir)
    CamFX.fov = { de = de, para = para, dur = dur, ini = os.clock(),
                  estilo = estilo or Enum.EasingStyle.Quad, dir = dir or Enum.EasingDirection.Out }
    camLigar()
end

local function camCongelar(cf)
    CamFX.congelar = cf
    if cf then camLigar() end
end

local function camCustom(fn)
    CamFX.custom = fn
    if fn then camLigar() end
end

local function pulso(efeito, prop, pico, dur)
    efeito.Enabled = true
    efeito[prop] = pico
    local meu = (efTokens[efeito] or 0) + 1
    efTokens[efeito] = meu
    tw(efeito, dur, { [prop] = 0 })
    task.delay(dur + 0.05, function()
        if efTokens[efeito] == meu and efeito.Parent then efeito.Enabled = false end
    end)
end

local function gradeMorte()
    efl.cc.Enabled = true
    tw(efl.cc, 0.5, { Saturation = -0.85, Contrast = 0.3, Brightness = -0.06, TintColor = Color3.fromRGB(255, 195, 215) })
end

local function resetGrade(dur)
    dur = dur or 0.5
    tw(efl.cc, dur, { Saturation = 0, Contrast = 0, Brightness = 0, TintColor = Color3.new(1, 1, 1) })
    task.delay(dur + 0.05, function()
        if not emMorte and efl.cc.Parent then efl.cc.Enabled = false end
    end)
end

local function gradeRespawn()
    efl.cc.Enabled = true
    efl.cc.Saturation, efl.cc.Contrast, efl.cc.Brightness = 0.9, 0.45, 0.12
    efl.cc.TintColor = Color3.fromRGB(190, 255, 255)

end

local function tocarSom(ids, vol, vel, fxs)
    if not CONFIG.Audio.Ativo or encerrado then return nil end
    if type(ids) == "string" then ids = { ids } end
    for _, id in ipairs(ids) do
        if id ~= "" and (audioFalhas[id] or 0) < (id:find("^rbxasset://") and 2 or 3) then
            local s = Instance.new("Sound")
            s.SoundId = id
            s.Volume = (vol or 1) * CONFIG.Audio.VolumeMestre
            s.PlaybackSpeed = vel or 1
            for _, e in ipairs(fxs or {}) do
                local ok, fx = pcall(Instance.new, e[1])
                if ok and fx then
                    for k, v in pairs(e[2]) do pcall(function() fx[k] = v end) end
                    fx.Parent = s
                end
            end
            s.Parent = SoundService
            local t0 = os.clock()
            local espera = id:find("^rbxasset://") and 0.4 or 3.0
            while not s.IsLoaded and os.clock() - t0 < espera and not encerrado do task.wait(0.05) end
            if s.IsLoaded then
                audiosAtivos[s] = true
                s.Ended:Connect(function() audiosAtivos[s] = nil; s:Destroy() end)
                task.delay(12, function() audiosAtivos[s] = nil; s:Destroy() end)
                s:Play()
                return s
            end
            audioFalhas[id] = (audioFalhas[id] or 0) + 1
            if CONFIG.DEBUG then warn("[Capri Audio] áudio não carregou ou não está autorizado: " .. tostring(id)) end
            s:Destroy()
        end
    end
    return nil
end

local function tocarVoz(chave, volume, velocidade)
    if not CONFIG.Audio.Ativo or not CONFIG.Audio.FalasCapri then return end
    local id = CAPRI_VOZ[chave]
    if not id then return end
    if (chave == "extra1" or chave == "extra2") and not CONFIG.Audio.AudiosDesconhecidos then return end
    seguro(tocarSom, id, volume or CONFIG.Audio.VolumeVoz, velocidade or rf(0.97, 1.03), {
        { "ReverbSoundEffect", { DecayTime = 0.75, Density = 0.65, Diffusion = 0.8, WetLevel = -8 } },
    })
end

local aberToken = 0
local function aberracao(dur, desloc, forca)
    aberToken = aberToken + 1
    local meu = aberToken
    local ini = os.clock()
    forca = forca or 0.45
    local conn
    conn = RunService.RenderStepped:Connect(function()
        local p = (os.clock() - ini) / dur
        if encerrado or meu ~= aberToken or p >= 1 then
            conn:Disconnect()
            if meu == aberToken then
                G.aberR.BackgroundTransparency, G.aberC.BackgroundTransparency = 1, 1
            end
            return
        end
        local k = 1 - p
        local d = desloc * k * (0.5 + math.random())
        G.aberR.Position = UDim2.new(0, -d, 0, 0)
        G.aberC.Position = UDim2.new(0, d, 0, 0)
        local t = 1 - (1 - forca) * k * (0.6 + 0.4 * math.random())
        G.aberR.BackgroundTransparency, G.aberC.BackgroundTransparency = t, t
    end)
end

local function novaParte(props)
    local p = Instance.new("Part")

    p.CastShadow = false

local function alturaChao(pos, ignorar)

    local lista = { fxFolder }

    return r and (r.Position.Y + 0.15) or (pos.Y - 3)
end

local function voxelDesintegrar(character)
    local partes = listarPartes(character, true)
    if #partes == 0 then return end

    task.spawn(function()
        local ini = os.clock()
        while os.clock() - ini < 0.9 and not encerrado do
            local a = (os.clock() - ini) / 0.9
            for _, p in ipairs(partes) do
                if p.Parent then
                    p.LocalTransparencyModifier = math.min(1, a * 1.3 + (math.random() < 0.2 and 0.25 or 0))
                end
            end
            task.wait(0.04)
        end
        for _, p in ipairs(partes) do
            if p.Parent then p.LocalTransparencyModifier = 1 end
        end
    end)

    for _ = 1, N(CONFIG.Morte.Voxels) do
        local base = pick(partes)
        local s = base.Size
        local off = Vector3.new(rf(-0.5, 0.5) * s.X, rf(-0.5, 0.5) * s.Y, rf(-0.5, 0.5) * s.Z)
        local pos = base.CFrame:PointToWorldSpace(off)
        local tam = rf(0.18, 0.5)
        local cubo = novaParte({ Size = Vector3.new(tam, tam, tam), CFrame = CFrame.new(pos),
                                 Color = pick(NEONS), Transparency = rf(0, 0.25) })
        local dir = Vector3.new(rf(-1, 1), rf(0.2, 1.4), rf(-1, 1))
        if dir.Magnitude < 0.01 then dir = Vector3.new(0, 1, 0) end
        local dest = pos + dir.Unit * rf(4, 13)
        local dur = rf(0.7, 1.5)
        tw(cubo, dur, {
            CFrame = CFrame.new(dest) * CFrame.Angles(rf(0, 6.28), rf(0, 6.28), rf(0, 6.28)),
            Size = Vector3.new(0.04, 0.04, 0.04), Transparency = 1,
        }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        task.delay(dur + 0.1, function() cubo:Destroy() end)
    end
end

end

local function piscarCorpo(character, dur)
    local partes, originais = {}, {}
    for _, d in ipairs(character:GetDescendants()) do
        if d:IsA("BasePart") then
            partes[#partes + 1] = d
            originais[d] = d.LocalTransparencyModifier
        end
    end
    local ini = os.clock()
    while os.clock() - ini < dur and character.Parent and not encerrado do
        local p = (os.clock() - ini) / dur
        local v = (math.random() < (1 - p) * 0.8) and rf(0.4, 1) or 0
        for _, part in ipairs(partes) do part.LocalTransparencyModifier = v end
        task.wait(0.04)
    end
    for _, part in ipairs(partes) do
        if part.Parent then part.LocalTransparencyModifier = originais[part] or 0 end
    end
end

local CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789#$%&@<>/\\|"
local revTokens = {}

local function revelar(labels, alvo, dur)
    local chave = labels[1]
    revTokens[chave] = (revTokens[chave] or 0) + 1
    local meu = revTokens[chave]
    task.spawn(function()
        local ini, n = os.clock(), #alvo
        while not encerrado and revTokens[chave] == meu do
            local p = (os.clock() - ini) / dur
            if p >= 1 then break end
            local fixos = math.floor(p * n)
            local partes = { alvo:sub(1, fixos) }
            for i = fixos + 1, n do
                local ch = alvo:sub(i, i)
                if ch == " " then
                    partes[#partes + 1] = " "
                else
                    local r = math.random(#CHARS)
                    partes[#partes + 1] = CHARS:sub(r, r)
                end
            end
            local s = table.concat(partes)
            for _, l in ipairs(labels) do l.Text = s end
            task.wait(0.03)
        end
        if revTokens[chave] == meu then
            for _, l in ipairs(labels) do l.Text = alvo end
        end
    end)
end

local function hexLinha(n)
    local t = {}
    for i = 1, n do
        t[i] = string.format("%04X %06X %02X", math.random(0, 65535), math.random(0, 16777215), math.random(0, 255))
    end
    return table.concat(t, "\n")
end

local function ecgValor(fase)
    if fase < 0.10 then return 0.12 * math.sin(fase / 0.10 * math.pi)
    elseif fase < 0.16 then return 0
    elseif fase < 0.19 then return -0.18
    elseif fase < 0.23 then return 1.0
    elseif fase < 0.27 then return -0.35
    elseif fase < 0.34 then return 0
    elseif fase < 0.50 then return 0.22 * math.sin((fase - 0.34) / 0.16 * math.pi)
    end
    return 0
end

    local rp = character:FindFirstChild("HumanoidRootPart")
    if not rp then return end
    local pos = rp.Position
    local chao = Vector3.new(pos.X, alturaChao(pos, character), pos.Z)
    local camera = workspace.CurrentCamera

    emMorte = false
    if hudConn then hudConn:Disconnect(); hudConn = nil end

    audioRespawn()

    if R.GradeDecrypt then

    if R.Bloom then pulso(efl.bloom, "Intensity", 3, 0.9) end
    if R.Blur then pulso(efl.blur, "Size", 16, 0.5) end
    if R.Aberracao then aberracao(0.7, 16, 0.55) end
    if R.Scanlines then
        G.scan.Visible = true
        task.delay(0.6, function() if not emMorte then G.scan.Visible = false end end)
    else
        G.scan.Visible = false
    end
    if R.Shake then camShake(1.6, 0.45, false) end
    if camera then
        if R.PunchFOV then
            camFov(fov0 + 40, fov0, 0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
        else
            camFov(camera.FieldOfView, fov0, 0.4)
        end

    if R.EstrelaFlare then estrelaFlare(PALETA.Ciano, 0.9) end
    for i = 1, R.Ondas do
        ondaDiamante((i % 2 == 0) and PALETA.Magenta or PALETA.Ciano, 0.7, 0.07 * (i - 1))
    end
    if R.Particulas then particulas2D(70, { PALETA.Ciano, PALETA.Magenta, PALETA.Amarelo, PALETA.Branco }) end

    if R.Voxels > 0 then seguro(voxelMaterializar, rp) end

    if R.PiscarCorpo then seguro(piscarCorpo, character, 0.7) end
    if R.Particulas then particulas3D(rp, 70, { PALETA.Verde, PALETA.Magenta, PALETA.Ciano }) end

    seguro(hudRespawn, ciclo)
end

local function limparFreeze()
    for _, c in ipairs(freezeConexoes) do pcall(function() c:Disconnect() end) end
    freezeConexoes = {}
    pcall(function() RunService:UnbindFromRenderStep(FREEZE_BIND) end)

local function iniciarFreeze(character, duracao, semCamera)
    local F = CONFIG.Freeze
    if not F.Ativo then return end
    freezeToken = freezeToken + 1
    local meu = freezeToken
    freezeAtivo = true
    limparFreeze()

    local hum = character:FindFirstChildOfClass("Humanoid")
    local rp = character:FindFirstChild("HumanoidRootPart")
    local camera = workspace.CurrentCamera
    if not (hum and rp and camera) then freezeAtivo = false; return end

    if F.Camera and not semCamera then camCongelar(camera.CFrame) end

    if F.Movimento or F.Pulo then
        RunService:BindToRenderStep(FREEZE_BIND, 250, function()
            if freezeAtivo and freezeToken == meu and hum.Parent then
                if F.Movimento then pcall(function() hum:Move(Vector3.zero, false) end) end
                if F.Pulo then hum.Jump = false end
            end
        end)
    end

    freezeConexoes[#freezeConexoes + 1] = RunService.Heartbeat:Connect(function()
        if freezeAtivo and freezeToken == meu and rp.Parent then
            rp.AssemblyLinearVelocity = Vector3.zero
            rp.AssemblyAngularVelocity = Vector3.zero
        end
    end)

    task.delay(duracao, function()
        if freezeToken ~= meu then return end
        freezeAtivo = false
        limparFreeze()
        camCongelar(nil)
    end)
end

local function obterMorphDesc()
    if morphDescCache then return morphDescCache end

        if okId and res then id = res end
    end

local function aplicarMorph()
    if not CONFIG.Morph.Ativo then return false end

    local character = player.Character
    if not character or not character.Parent then return false end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end

    -- ⛔ BLOQUEIO R6 (com aviso único)
    if humanoid.RigType ~= Enum.HumanoidRigType.R6 then
        if not avisoR15JaMostrado then
            avisoR15JaMostrado = true
            if CONFIG.DEBUG then
                warn("[Capri] 🚫 Morph bloqueado: personagem é R15 (só funciona em R6)")
            end
        end
        return false
    end

    local desc = obterMorphDesc()
    if not desc then return false end

    -- Força R6 (garantido pelo bloqueio acima)
    local okM, gen = pcall(function()
        return Players:CreateHumanoidModelFromDescription(desc, Enum.HumanoidRigType.R6)
    end)
    if not okM or not gen then return false end

    for _, item in ipairs(character:GetChildren()) do
        if item:IsA("Accessory") or item:IsA("Clothing") or item:IsA("ShirtGraphic")
        or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
            pcall(function() item:Destroy() end)
        end
    end

    for _, item in ipairs(gen:GetChildren()) do
        local ex = character:FindFirstChild(item.Name)
        if ex and ex:IsA("BasePart") and ex.Name ~= "HumanoidRootPart" then
            pcall(function() ex.Color = item.Color end)
        end
    end

    local tHead, cHead = gen:FindFirstChild("Head"), character:FindFirstChild("Head")
    if tHead and cHead then
        local oldF = cHead:FindFirstChildOfClass("Decal"); if oldF then oldF:Destroy() end
        local nF = tHead:FindFirstChildOfClass("Decal")
        if nF then
            nF:Clone().Parent = cHead
        else
            local def = Instance.new("Decal"); def.Name = "face"
            def.Texture = "rbxasset://textures/face.png"; def.Parent = cHead
        end
        local oldM = cHead:FindFirstChildOfClass("SpecialMesh"); if oldM then oldM:Destroy() end
        local nM = tHead:FindFirstChildOfClass("SpecialMesh"); if nM then nM:Clone().Parent = cHead end
    end

    for _, item in ipairs(gen:GetChildren()) do
        if item:IsA("Clothing") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
            pcall(function() item:Clone().Parent = character end)
        elseif item:IsA("Accessory") then
            local cl = item:Clone()
            local h = cl:FindFirstChild("Handle")
            if h then
                local att = h:FindFirstChildOfClass("Attachment")
                if att then
                    local tAtt = character:FindFirstChild(att.Name, true)
                    if tAtt then
                        h.CFrame = tAtt.WorldCFrame
                        local w = Instance.new("Weld")
                        w.Name = "AccessoryWeld"
                        w.Part0, w.Part1 = h, tAtt.Parent
                        w.C0, w.C1 = att.CFrame, tAtt.CFrame
                        w.Parent = h
                    end
                end
            end
            pcall(function() cl.Parent = character end)
        end
    end
    gen:Destroy()
    return true
end

-- controla a transparência local do corpo (flicker / sumir / voltar)
local function capturarCorpo(character)
    local partes, orig = {}, {}
    for _, d in ipairs(character:GetDescendants()) do
        if d:IsA("BasePart") then
            partes[#partes + 1] = d
            orig[d] = d.LocalTransparencyModifier
        end
    end
    local c = {}
    function c.aplicar(v)
        for _, p in ipairs(partes) do
            if p.Parent then p.LocalTransparencyModifier = v end
        end
    end
    function c.restaurar()
        for _, p in ipairs(partes) do
            if p.Parent then p.LocalTransparencyModifier = orig[p] or 0 end
        end
    end
    return c
end

-- explosão de voxels neon num ponto
local function voxelBurst(pos, qtd, raioMin, raioMax)
    for _ = 1, N(qtd) do
        local dir = Vector3.new(rf(-1, 1), rf(-0.2, 1.2), rf(-1, 1))
        if dir.Magnitude < 0.01 then dir = Vector3.new(0, 1, 0) end
        local ini = pos + Vector3.new(rf(-1, 1), rf(-2, 2), rf(-1, 1))
        local tam = rf(0.18, 0.5)
        local cubo = novaParte({ Size = Vector3.new(tam, tam, tam), CFrame = CFrame.new(ini),
                                 Color = pick(NEONS), Transparency = rf(0, 0.2) })
        local dur = rf(0.45, 1.0)
        tw(cubo, dur, {
            CFrame = CFrame.new(ini + dir.Unit * rf(raioMin or 5, raioMax or 14)) * CFrame.Angles(rf(0, 6.28), rf(0, 6.28), 0),
            Size = Vector3.new(0.04, 0.04, 0.04), Transparency = 1,
        }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        task.delay(dur + 0.1, function() cubo:Destroy() end)
    end
end

-- true/false se já sabemos; se ainda não, agenda a checagem e devolve false por enquanto
function LISTA.ehAmigo(p)
    local id = p.UserId
    local v = LISTA.amigo[id]
    if v == nil and not LISTA.naFila[id] then
        LISTA.naFila[id] = true
        LISTA.fila[#LISTA.fila + 1] = id
        LISTA.processar()
    end
    return v == true
end

-- checa um por vez (evita estourar o limite da API de amizades)
function LISTA.processar()
    if LISTA.rodando then return end
    LISTA.rodando = true
    task.spawn(function()
        while not encerrado and #LISTA.fila > 0 do
            local id = table.remove(LISTA.fila, 1)
            local ok, r = pcall(function() return player:IsFriendsWith(id) end)
            if ok then
                LISTA.amigo[id] = (r == true)
                LISTA.naFila[id] = nil
                if r == true and LISTA.aoMudar then LISTA.aoMudar() end
            else
                local n = (LISTA.tentativas[id] or 0) + 1
                LISTA.tentativas[id] = n
                if n < 3 then
                    task.delay(1.5, function()
                        if encerrado then return end
                        LISTA.fila[#LISTA.fila + 1] = id
                        LISTA.processar()
                    end)
                else
                    LISTA.amigo[id] = false
                    LISTA.naFila[id] = nil
                end
            end
            task.wait(0.12)
        end
        LISTA.rodando = false
    end)
end

function LISTA.carregar()
    if not CONFIG.Lista.Salvar then return end
    if not (isfile and readfile) then return end
    local ok, dados = pcall(function()
        if isfile(CONFIG.Lista.Arquivo) then
            return HttpService:JSONDecode(readfile(CONFIG.Lista.Arquivo))
        end
        return nil
    end)
    if ok and type(dados) == "table" then
        for _, r in ipairs(dados) do
            if type(r) == "table" and tonumber(r.userId) and type(r.name) == "string" then
                LISTA.recentes[#LISTA.recentes + 1] = {
                    userId = tonumber(r.userId), name = r.name, display = tostring(r.display or r.name),
                }
                if #LISTA.recentes >= CONFIG.Lista.Recentes then break end
            end
        end
    end
end

function LISTA.salvar()
    if not CONFIG.Lista.Salvar or not writefile then return end
    pcall(function()
        writefile(CONFIG.Lista.Arquivo, HttpService:JSONEncode(LISTA.recentes))
    end)
end

-- chamado quando um TP dá certo: o alvo vai para o topo dos recentes
function LISTA.registrar(p)
    for i = #LISTA.recentes, 1, -1 do
        if LISTA.recentes[i].userId == p.UserId then table.remove(LISTA.recentes, i) end
    end
    table.insert(LISTA.recentes, 1, { userId = p.UserId, name = p.Name, display = p.DisplayName })
    while #LISTA.recentes > CONFIG.Lista.Recentes do table.remove(LISTA.recentes) end
    LISTA.salvar()
    if LISTA.aoMudar then LISTA.aoMudar() end
end

function LISTA.posicaoRecente(userId)   -- 1 = mais recente
    for i, r in ipairs(LISTA.recentes) do
        if r.userId == userId then return i end
    end
    return nil
end

-- devolve (amigos, recentes) presentes no servidor; AMIGOS SEMPRE PRIMEIRO
function LISTA.montar()
    local amigos, recs = {}, {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local ri = LISTA.posicaoRecente(p.UserId)
            if LISTA.ehAmigo(p) then
                amigos[#amigos + 1] = { player = p, rec = ri or 1e9 }
            elseif ri then
                recs[#recs + 1] = { player = p, rec = ri }
            end
        end
    end
    table.sort(amigos, function(a, b)
        if a.rec ~= b.rec then return a.rec < b.rec end   -- amigo que você visitou há pouco vem antes
        return a.player.DisplayName:lower() < b.player.DisplayName:lower()
    end)
    table.sort(recs, function(a, b) return a.rec < b.rec end)
    return amigos, recs
end

local tpCooldownAte = 0
local tpOcupado = false

local function posicaoAtras(aChar, dist)
    local tRoot = aChar:FindFirstChild("HumanoidRootPart")
    if not tRoot then return nil end
    local base = tRoot.CFrame
    local look = Vector3.new(base.LookVector.X, 0, base.LookVector.Z)
    if look.Magnitude < 0.05 then look = Vector3.new(0, 0, -1) end
    look = look.Unit
    local origem = base.Position
    local desejado = origem - look * dist

    local ignorar = { aChar, fxFolder }
    if player.Character then ignorar[#ignorar + 1] = player.Character end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignorar
    local dir = desejado - origem
    local r = workspace:Raycast(origem, dir, params)
    if r then desejado = r.Position - dir.Unit * 1.2 end   -- parede atrás do alvo: fica logo na frente dela

    return CFrame.lookAt(desejado, desejado + look)         -- olhando na mesma direção do alvo (para as costas dele)
end

local function teleportarAtras(alvo)
    if encerrado or tpOcupado then return false end
    local t0 = os.clock()
    if t0 < tpCooldownAte then return false end

    local meuChar = player.Character
    local meuRoot = meuChar and meuChar:FindFirstChild("HumanoidRootPart")
    local hum = meuChar and meuChar:FindFirstChildOfClass("Humanoid")
    if not (meuRoot and hum) or hum.Health <= 0 then
        toast("SIGNAL LOST // CANNOT LINK", PALETA.Vermelho)
        sfx.erro()
        return false
    end

    local aChar = alvo and alvo.Character
    local aRoot = aChar and aChar:FindFirstChild("HumanoidRootPart")
    if not aRoot then
        toast("TARGET OFFLINE // NO CONSTRUCT", PALETA.Vermelho)
        sfx.erro()
        return false
    end

    local destino = posicaoAtras(aChar, CONFIG.TP.Distancia)
    if not destino then return false end

    tpOcupado = true
    tpCooldownAte = t0 + CONFIG.TP.Cooldown + 0.9
    local corpo = capturarCorpo(meuChar)

    local ok, err = pcall(function()
        local camera = workspace.CurrentCamera
        local fov0 = camera and camera.FieldOfView or 70
        local origem = meuRoot.Position
        local E = CONFIG.TP.Efeitos

        -- SAÍDA
        if E then
            sfx.tpSaida()
            flashTela(PALETA.Ciano, 0.25, 0.25)
            aberracao(0.45, 14, 0.5)
            glitchBlocos(0.3, 12)
            voxelBurst(origem, 50, 5, 14)
            discoChao(Vector3.new(origem.X, alturaChao(origem, meuChar), origem.Z), PALETA.Magenta, 12, 0.6, 0)
            luzFlash(origem, PALETA.Magenta, 0.5)
            feixeSatelite(origem, 120)
            if camera then camFov(camera.FieldOfView, fov0 - 14, 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In) end
            camShake(1.4, 0.25, true)
            local ini = os.clock()
            while os.clock() - ini < 0.2 and meuChar.Parent do
                corpo.aplicar(math.min(1, (os.clock() - ini) / 0.2 * 1.2 + (math.random() < 0.3 and 0.2 or 0)))
                task.wait(0.03)
            end
            corpo.aplicar(1)
        end

        -- TELEPORTE (recalcula: o alvo pode ter andado)
        if not (meuChar.Parent and aChar.Parent and aRoot.Parent) then return end
        destino = posicaoAtras(aChar, CONFIG.TP.Distancia) or destino
        local okPivot = pcall(function() meuChar:PivotTo(destino) end)
        if not okPivot then meuRoot.CFrame = destino end
        meuRoot.AssemblyLinearVelocity = Vector3.zero
        meuRoot.AssemblyAngularVelocity = Vector3.zero

        -- CHEGADA
        local pos = destino.Position
        if E then
            sfx.tpChegada()
            flashTela(PALETA.Branco, 0.3, 0.1)
            local chao = Vector3.new(pos.X, alturaChao(pos, meuChar), pos.Z)
            voxelMaterializar(meuRoot, 55)
            feixeSatelite(chao, 160)
            discoChao(chao, PALETA.Ciano, 16, 0.8, 0.05)
            discoChao(chao, PALETA.Magenta, 22, 0.9, 0.15)
            luzFlash(pos, PALETA.Ciano, 0.8)
            estrelaFlare(PALETA.Magenta, 0.7)
            ondaDiamante(PALETA.Ciano, 0.6, 0)
            ondaDiamante(PALETA.Magenta, 0.6, 0.08)
            particulas3D(meuRoot, 60, { PALETA.Ciano, PALETA.Magenta, PALETA.Amarelo })
            pulso(efl.bloom, "Intensity", 2.4, 0.7)
            aberracao(0.55, 14, 0.5)
            camShake(1.8, 0.4, false)
            if camera then camFov(fov0 + 30, fov0, 0.7, Enum.EasingStyle.Quint, Enum.EasingDirection.Out) end
            seguro(highlightGlitch, meuChar, 0.9)
            local ini = os.clock()
            while os.clock() - ini < 0.6 and meuChar.Parent do
                local p = (os.clock() - ini) / 0.6
                corpo.aplicar((math.random() < (1 - p) * 0.8) and rf(0.4, 1) or 0)
                task.wait(0.04)
            end
        end

        LISTA.registrar(alvo)
        toast("LINK ESTABLISHED // " .. (ehAscii(alvo.DisplayName) and string.upper(alvo.DisplayName) or alvo.DisplayName)
              .. " (@" .. alvo.Name .. ")", PALETA.Ciano)
    end)

    corpo.restaurar()
    tpOcupado = false
    if not ok then warn("[Capri TP] erro:", err) end
    return ok
end

        local vw, vh = viewport()

        local vw, vh = viewport()
        cx = math.clamp(cx, w / 2 + 2, math.max(w / 2 + 2, vw - w / 2 - 2))
        cy = math.clamp(cy, h / 2 + 2, math.max(h / 2 + 2, vh - h / 2 - 2))

    end

        local p = {

            AutoButtonColor = false, Font = Enum.Font.Code, TextSize = 13, TextColor3 = PALETA.Ciano, Text = "",

        return novo("TextButton", p, pai)
    end

        rastrear(b.MouseEnter:Connect(function()
            if not semSom then sfx.hover() end
            tw(b, 0.1, { BackgroundTransparency = 0 })
        end))
        rastrear(b.MouseLeave:Connect(function() tw(b, 0.15, { BackgroundTransparency = 0.15 }) end))

    local function tornarArrastavel(alvoInput, mover, aoClicar)
        local arrastando, moveu, dragInput, inicioPos, inicioObj = false, false, nil, nil, nil
        rastrear(alvoInput.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                arrastando, moveu = true, false
                dragInput = input
                inicioPos = input.Position
                inicioObj = Vector2.new(mover.Position.X.Offset, mover.Position.Y.Offset)
                local conn
                conn = input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        conn:Disconnect()
                        arrastando = false
                        if not moveu and aoClicar then aoClicar() end
                    end
                end)
            end
        end))
        rastrear(UserInputService.InputChanged:Connect(function(input)
            if not arrastando then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement or input == dragInput then
                local d = input.Position - inicioPos
                if not moveu and d.Magnitude < 6 then return end
                moveu = true
                local tam = mover.AbsoluteSize
                local cx, cy = limitarPos(inicioObj.X + d.X, inicioObj.Y + d.Y, tam.X, tam.Y)

    -- ---- busca de nick (Original = Name | Visual = DisplayName); AMIGOS sempre primeiro ----
    local function buscarAlvos(txt)
        local q = txt:lower()
        local lista = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player then
                local n, d = p.Name, p.DisplayName
                local nl, dl = n:lower(), d:lower()
                local melhor = nil
                if nl:sub(1, #q) == q then
                    melhor = { player = p, texto = n, campo = "ORIGINAL", prefixo = true }
                end
                if dl:sub(1, #q) == q then
                    if not melhor or #d < #n then
                        melhor = { player = p, texto = d, campo = "VISUAL", prefixo = true }
                    end
                end
                if not melhor then
                    local achouN = nl:find(q, 1, true)
                    local achouD = dl:find(q, 1, true)
                    if achouN or achouD then
                        melhor = { player = p, texto = achouN and n or d, campo = achouN and "ORIGINAL" or "VISUAL", prefixo = false }
                    end
                end
                if melhor then
                    melhor.amigo = LISTA.ehAmigo(p)
                    melhor.rec = LISTA.posicaoRecente(p.UserId) or 1e9
                    lista[#lista + 1] = melhor
                end
            end
        end
        table.sort(lista, function(a, b)
            if a.prefixo ~= b.prefixo then return a.prefixo end
            if a.amigo ~= b.amigo then return a.amigo end
            if a.rec ~= b.rec then return a.rec < b.rec end
            if #a.texto ~= #b.texto then return #a.texto < #b.texto end
            return a.texto:lower() < b.texto:lower()
        end)
        return lista
    end

            MN.ghost.Text = string.rep(" ", #bruto) .. melhor.texto:sub(#txt + 1)
        else
            MN.ghost.Text = ""

        atualizarChips(lista)

            MN.ultimoAlvo = melhor.player

    function atualizarLista()
        if not MN.listaScroll then return end
        limparLista()
        local amigos, recs = LISTA.montar()
        local max = CONFIG.Lista.Maximo
        local ordem, usados = 0, 0

        if #amigos > 0 then
            ordem = ordem + 1
            criarCabecalhoLista(ordem, "-- FRIENDS (" .. #amigos .. ") --", PALETA.Ouro)
            for _, e in ipairs(amigos) do
                if usados >= max then break end
                ordem = ordem + 1
                usados = usados + 1
                criarLinhaLista(ordem, e.player, "amigo")
            end
        end
        if #recs > 0 and usados < max then
            ordem = ordem + 1
            criarCabecalhoLista(ordem, "-- RECENT TARGETS --", PALETA.Magenta)
            for _, e in ipairs(recs) do
                if usados >= max then break end
                ordem = ordem + 1
                usados = usados + 1
                criarLinhaLista(ordem, e.player, "recente")
            end
        end
        if ordem == 0 then
            local l = texto({
                Size = UDim2.new(1, -6, 0, 48), Font = Enum.Font.Code, TextSize = 11, TextWrapped = true,
                TextColor3 = Color3.fromRGB(140, 150, 180), LayoutOrder = 1, ZIndex = ZM + 5,
                Text = "NO FRIENDS OR RECENT TARGETS IN THIS SERVER YET.\nTELEPORT TO SOMEONE AND THEY SHOW UP HERE.",
            }, MN.listaScroll)
            MN.linhasLista[#MN.linhasLista + 1] = l
        end
    end

    function trocarAba(nome, semSom)
        if not MN.pags[nome] then return end
        if not semSom and MN.aba ~= nome then sfx.tick() end
        MN.aba = nome
        for k, pg in pairs(MN.pags) do pg.Visible = (k == nome) end
        if nome ~= "tp" then pcall(function() MN.caixa:ReleaseFocus() end) end
        estilizarAbas()
        if nome == "lista" then atualizarLista() end
        if nome == "tp" then atualizarBusca() end
    end

    -- ---- loop visual do menu (nick com efeitos, varredura, borda) ----
    local function menuLoopIniciar()
        if MN.loopConn then return end
        local ini = os.clock()
        local ult = 0
        local proxScramble = ini + rf(3, 6)
        local nomeV = player.DisplayName
        local ascii = ehAscii(nomeV)
        MN.loopConn = RunService.RenderStepped:Connect(function()
            if encerrado or not MN.aberto then
                if MN.loopConn then MN.loopConn:Disconnect(); MN.loopConn = nil end
                return
            end
            local t = os.clock()
            MN.barraScan.Position = UDim2.new(0, 0, 0, ((t - ini) * 150) % (BH + 90) - 60)
            if t - ult < 0.05 then return end
            ult = t

            local k = (math.sin(t * 2.2) + 1) / 2
            MN.nickStroke.Color = PALETA.Ciano:Lerp(PALETA.Magenta, k)

            if math.random() < 0.08 then
                local d = math.random(2, 8)
                MN.nickR.Position = UDim2.new(0, 14 - d, 0, 60 + math.random(-2, 2))
                MN.nickC.Position = UDim2.new(0, 14 + d, 0, 60 + math.random(-2, 2))
                if ascii and math.random() < 0.5 then
                    local emb = embaralhar(nomeV, 0.35)
                    MN.nick.Text, MN.nickR.Text, MN.nickC.Text = emb, emb, emb
                    task.delay(0.07, function()
                        if MN.nick.Parent then MN.nick.Text, MN.nickR.Text, MN.nickC.Text = nomeV, nomeV, nomeV end
                    end)
                end
            else
                MN.nickR.Position = UDim2.new(0, 12, 0, 60)
                MN.nickC.Position = UDim2.new(0, 16, 0, 60)
            end

            if t > proxScramble then
                proxScramble = t + rf(4, 8)
                if ascii then revelar({ MN.nick, MN.nickR, MN.nickC }, nomeV, 0.45) end
            end

            if math.random() < 0.05 then
                MN.rootStroke.Color = pick({ PALETA.Magenta, PALETA.Ouro, PALETA.Ciano })
            else
                MN.rootStroke.Color = PALETA.Ciano
            end
        end)
    end

                MN.root.Visible = false; task.wait(0.025)
                MN.root.Visible = true;  task.wait(0.03)
            end
            if MN.animToken ~= meu then return end
            tw(MN.root, 0.3, { Size = UDim2.fromOffset(W, H) }, Enum.EasingStyle.Quart)
            tw(MN.corpo, 0.3, { BackgroundTransparency = 0.12 })
            task.wait(0.12)

                revelar({ MN.nick, MN.nickR, MN.nickC }, nomeV, 0.55)

            end
            if ehAscii(player.Name) then revelar({ MN.valOriginal }, player.Name, 0.45) end
        end)
        menuLoopIniciar()
        atualizarFooter()
        atualizarAuraUI()
        if MN.aba == "lista" then atualizarLista() elseif MN.aba == "tp" then atualizarBusca() end
    end

    function fecharMenu()
        if not MN.aberto then return end
        MN.aberto = false
        MN.animToken = MN.animToken + 1
        local meu = MN.animToken
        if CONFIG.Menu.Som then sfx.fechar() end
        pcall(function() MN.caixa:ReleaseFocus() end)
        local W = tamanhoMenu()
        seguro(function()
            for _ = 1, 2 do
                if MN.animToken ~= meu then return end
                MN.root.Visible = false; task.wait(0.02)
                MN.root.Visible = true;  task.wait(0.025)
            end
            if MN.animToken ~= meu then return end
            tw(MN.root, 0.2, { Size = UDim2.fromOffset(W, 2) }, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
            task.wait(0.2)
            if MN.animToken ~= meu then return end
            tw(MN.root, 0.12, { Size = UDim2.fromOffset(0, 2) })
            task.wait(0.12)
            if MN.animToken == meu then MN.root.Visible = false end
        end)
    end

    function alternarMenu()
        if MN.aberto then fecharMenu() else abrirMenu() end
    end

    -- a lista mudou (amigo detectado, novo recente, alguém entrou/saiu): atualiza com debounce
    LISTA.aoMudar = function()
        if MN.listaSujo then return end
        MN.listaSujo = true
        task.delay(0.2, function()
            MN.listaSujo = false
            if encerrado or not MN.aberto then return end
            if MN.aba == "lista" then atualizarLista() elseif MN.aba == "tp" then atualizarBusca() end
        end)
    end

    if CONFIG.Menu.Ativo then
        local okM, errM = pcall(criarMenu)
        if not okM then
            warn("[Capri] o menu falhou ao montar (o resto continua funcionando):", errM)

            pcall(function() if MN.root then MN.root:Destroy() end end)
            MN.botao, MN.root = nil, nil
        end
    end
end

local function criarCamIntro(character, dur)
    local camera = workspace.CurrentCamera
    local head = character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    local root = character:FindFirstChild("HumanoidRootPart")
    if not (camera and head and root) then return nil end

    local foco0 = head.Position
    local offFim = camera.CFrame.Position - foco0
    local distFim = offFim.Magnitude
    if distFim < 3 then return nil end            -- 1ª pessoa: sem cinemática

    local lookFim = camera.CFrame.LookVector
    local fovFim = camera.FieldOfView
    local yawFim = math.atan2(offFim.X, offFim.Z)
    local pitchFim = math.asin(math.clamp(offFim.Y / distFim, -0.99, 0.99))
    local look = root.CFrame.LookVector
    local yawIni = math.atan2(look.X, look.Z)    -- começa NA FRENTE do personagem
    local delta = ((yawFim - yawIni + math.pi) % (2 * math.pi)) - math.pi
    local ini = os.clock()

    return function(c)
        local p = (os.clock() - ini) / dur
        local a = math.min(1, math.max(0, p))
        local e = TweenService:GetValue(a, Enum.EasingStyle.Quart, Enum.EasingDirection.InOut)
        local yaw = yawIni + delta * e
        local pitch = 0.1 + (pitchFim - 0.1) * e
        local dist = 6.5 + (distFim - 6.5) * e
        local foco = head.Position
        local dirV = Vector3.new(math.sin(yaw) * math.cos(pitch), math.sin(pitch), math.cos(yaw) * math.cos(pitch))
        local pos = foco + dirV * dist
        local dir = (foco - pos).Unit:Lerp(lookFim, e * e * e)
        c.CFrame = CFrame.lookAt(pos, pos + dir)
        c.Focus = CFrame.new(foco)
        c.FieldOfView = 46 + (fovFim - 46) * e
        if p >= 1 then CamFX.custom = nil end
    end
end

local function alturaVoid() return workspace.FallenPartsDestroyHeight + 60 end

local function atualizarMarcador(cf)
    if not marcador or not marcador.Parent then
        marcador = Instance.new("Part")
        marcador.Name = "LocalInvisivel_Respawn"
        marcador.Size = Vector3.new(1, 1, 1)
        marcador.Anchored = true
        marcador.CanCollide, marcador.CanTouch, marcador.CanQuery = false, false, false
        marcador.Transparency = 1
        marcador.Massless = true
        marcador.Parent = workspace
    end
    marcador.CFrame = cf
end

local function endireitar(cf, reserva)
    local pos = cf.Position
    local look = cf.LookVector
    local plano = Vector3.new(look.X, 0, look.Z)
    if plano.Magnitude < 0.05 and reserva then
        local l2 = reserva.LookVector
        plano = Vector3.new(l2.X, 0, l2.Z)
    end
    if plano.Magnitude < 0.05 then return CFrame.new(pos) end
    return CFrame.lookAt(pos, pos + plano.Unit)
end

rastrear(RunService.Heartbeat:Connect(function()
    if not salvandoAtivo then return end
    local c = player.Character; if not c then return end
    local h = c:FindFirstChildOfClass("Humanoid")
    local rp = c:FindFirstChild("HumanoidRootPart")
    if not (h and rp) or h.Health <= 0 then return end

    local cf = rp.CFrame
    if cf.Position.Y < alturaVoid() then return end

    if h.FloorMaterial ~= Enum.Material.Air then cframeSeguro = cf end

    local t = os.clock()
    if t - tempoUltimoSalvamento < 1 and ultimoCFrameSalvo
    and (cf.Position - ultimoCFrameSalvo.Position).Magnitude <= 5 then return end

    cframeSalvo = cf
    atualizarMarcador(cf)
    ultimoCFrameSalvo = cf
    tempoUltimoSalvamento = t
end))

local function limparConexoes()
    for _, c in ipairs(conexoesCharacter) do pcall(function() c:Disconnect() end) end
    conexoesCharacter = {}
end

local function configurarCharacter(character)
    limparConexoes()
    local h = character:WaitForChild("Humanoid", 10)
    if not h then return end

    local healthAnterior = h.Health
    conexoesCharacter[#conexoesCharacter + 1] = h.HealthChanged:Connect(function(saudeAtual)
        if saudeAtual < healthAnterior and saudeAtual > 0 and CONFIG.Audio.ReacaoImpacto and capriOnImpact then
            capriOnImpact(character)
        end
        healthAnterior = saudeAtual
    end)

    conexoesCharacter[#conexoesCharacter + 1] = h.StateChanged:Connect(function(_, novoEstado)
        if novoEstado == Enum.HumanoidStateType.Jumping and CONFIG.Audio.ReacaoImpacto and capriOnDodge then
            capriOnDodge(character)
        end
    end)

    conexoesCharacter[#conexoesCharacter + 1] = h.Died:Connect(function()
        local rp = character:FindFirstChild("HumanoidRootPart")
        if rp and rp.Position.Y > alturaVoid() then
            cframeSalvo = endireitar(rp.CFrame, ultimoCFrameSalvo)
            atualizarMarcador(cframeSalvo)
        elseif cframeSeguro then
            cframeSalvo = cframeSeguro
        end
        salvandoAtivo = false
        seguro(sequenciaMorte, character, cicloAtual)
    end)
end

local function levarPara(character, rp, cf)
    local ok = pcall(function() character:PivotTo(cf) end)
    if not ok then pcall(function() rp.CFrame = cf end) end
    pcall(function()
        rp.AssemblyLinearVelocity = Vector3.zero
        rp.AssemblyAngularVelocity = Vector3.zero
    end)
end

-- true se o jogo te colocou longe do lugar escolhido (ignora só a queda vertical normal)
local function foraDoLugar(rp, alvoCF)
    local d = rp.Position - alvoCF.Position
    return Vector3.new(d.X, 0, d.Z).Magnitude > 6 or math.abs(d.Y) > 50
end

-- por ~0.5s confere se o jogo te moveu (spawn dele) e leva de volta
local function reforcarPosicao(character, hum, rp, alvoCF, ciclo)
    local fim = os.clock() + 0.5
    while os.clock() < fim and not encerrado and cicloAtual == ciclo and character.Parent and hum.Health > 0 do
        task.wait(0.05)
        if rp.Parent and foraDoLugar(rp, alvoCF) then levarPara(character, rp, alvoCF) end
    end
end

rastrear(player.CharacterAdded:Connect(function(character)
    if encerrado then return end
    salvandoAtivo = false
    cicloAtual = cicloAtual + 1
    local ciclo = cicloAtual

    -- 1º spawn (script executado antes de existir personagem): toca a intro no lugar do respawn
    if introEstado == "pendente" then
        seguro(configurarCharacter, character)
        seguro(iniciarIntroSegura, character)
        salvandoAtivo = true
        return
    end

    -- cobre a tela já (esconde o ponto de spawn padrão)
    tw(G.escuro, 0.08, { BackgroundTransparency = 0.05 })

    seguro(configurarCharacter, character)
    seguro(iniciarFreeze, character, CONFIG.Freeze.Duracao)

    local h = character:WaitForChild("Humanoid", 10)
    local rp = character:WaitForChild("HumanoidRootPart", 10)
    if not (h and rp) then salvandoAtivo = true; return end

    local t0 = os.clock()
    while h.Health <= 0 and os.clock() - t0 < 3 do task.wait() end

    -- teleporta para o lugar salvo N vezes (padrão: 2x, a cada 0.04s): vence o spawn que o jogo colocou
    local R = CONFIG.RespawnTP
    local alvoCF = nil
    if cframeSalvo then
        alvoCF = CFrame.new(cframeSalvo.Position + Vector3.new(0, CONFIG.OffsetY, 0))
                 * (cframeSalvo - cframeSalvo.Position)
        for _ = 1, R.Repeticoes do
            task.wait(R.Intervalo)
            if not character.Parent then break end
            levarPara(character, rp, alvoCF)
        end
    end
    local resto = 0.15 - (alvoCF and (R.Repeticoes * R.Intervalo) or 0)
    if resto > 0 then task.wait(resto) end
    if alvoCF and rp.Parent and foraDoLugar(rp, alvoCF) then levarPara(character, rp, alvoCF) end

    seguro(sequenciaRespawn, character, ciclo)

    -- 🎭 KinxCaesar: reaplica morph a cada respawn
    seguro(function() task.wait(0.3); aplicarMorph() end)
    if alvoCF and R.Reforco then seguro(reforcarPosicao, character, h, rp, alvoCF, ciclo) end

    task.delay(1.15, function()
        if not encerrado and capriIniciarAura then capriIniciarAura(character) end
    end)

    task.wait((alvoCF and R.Reforco) and 0.55 or 0.2)   -- espera o reforço terminar antes de religar o salvamento
    cframeSalvo, ultimoCFrameSalvo = nil, nil
    tempoUltimoSalvamento = os.clock()
    if h.Health > 0 then salvandoAtivo = true end
end))

local function capriObterPersonagem()
    if encerrado or emMorte or tpOcupado or introEstado == "rodando" then return nil end
    local character = player.Character
    if not character or not character.Parent then return nil end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or humanoid.Health <= 0 or not root then return nil end
    if humanoid.RigType ~= Enum.HumanoidRigType.R6 then return nil end

    return character, root
end

-- Aura persistente: contorno vermelho/rosa, partículas e fragmentos verdes orbitais.
-- A atualização é só visual e acompanha o RootPart local do personagem R6.
local CAPRI_AURA_STATE = { character = nil, folder = nil, connection = nil, shards = {}, attachments = {}, light = nil, highlight = nil }

capriLimparAura = function()
    if CAPRI_AURA_STATE.connection then
        pcall(function() CAPRI_AURA_STATE.connection:Disconnect() end)
        CAPRI_AURA_STATE.connection = nil
    end
    for _, attachment in ipairs(CAPRI_AURA_STATE.attachments) do
        if attachment and attachment.Parent then pcall(function() attachment:Destroy() end) end
    end
    CAPRI_AURA_STATE.attachments = {}
    if CAPRI_AURA_STATE.folder then
        pcall(function() CAPRI_AURA_STATE.folder:Destroy() end)
        CAPRI_AURA_STATE.folder = nil
    end
    CAPRI_AURA_STATE.character = nil
    CAPRI_AURA_STATE.shards = {}
    CAPRI_AURA_STATE.light = nil
    CAPRI_AURA_STATE.highlight = nil
end

local function capriCriarParticulaAura(parent, cores, taxa, tamanho, velocidade)
    local att = Instance.new("Attachment")
    att.Name = "Capri_AuraAttachment"
    att.Parent = parent
    local emitter = Instance.new("ParticleEmitter")
    emitter.Name = "Capri_AuraParticles"
    emitter.Texture = "rbxasset://textures/particles/sparkles_main.dds"
    emitter.Enabled = true
    emitter.Rate = taxa
    emitter.Lifetime = NumberRange.new(0.45, 0.95)
    emitter.Speed = NumberRange.new(velocidade * 0.35, velocidade)
    emitter.SpreadAngle = Vector2.new(180, 180)
    emitter.Drag = 2
    emitter.LightEmission = 1
    emitter.Rotation = NumberRange.new(0, 360)
    emitter.RotSpeed = NumberRange.new(-120, 120)
    emitter.Size = NumberSequence.new({
        NumberSequenceKeypoint.new(0, tamanho),
        NumberSequenceKeypoint.new(0.55, tamanho * 0.55),
        NumberSequenceKeypoint.new(1, 0),
    })
    emitter.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.12),
        NumberSequenceKeypoint.new(0.75, 0.35),
        NumberSequenceKeypoint.new(1, 1),
    })
    emitter.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, cores[1]),
        ColorSequenceKeypoint.new(0.55, cores[2]),
        ColorSequenceKeypoint.new(1, cores[3] or cores[1]),
    })
    emitter.Parent = att
    return att
end

local function capriCriarFragmentoAura(pai, indice, total)
    local shard = Instance.new("WedgePart")
    shard.Name = "Capri_OrbitShard_" .. tostring(indice)
    shard.Anchored = true
    shard.CanCollide, shard.CanTouch, shard.CanQuery = false, false, false
    shard.CastShadow = false
    shard.Material = Enum.Material.Neon
    shard.Size = Vector3.new(rf(0.22, 0.42), rf(0.5, 0.95), rf(0.2, 0.38))
    shard.Color = (indice % 4 == 0) and PALETA.Magenta or PALETA.Verde
    shard.Transparency = 0.06
    shard.Parent = pai

    local a0 = Instance.new("Attachment")
    a0.Position = Vector3.new(0, -shard.Size.Y * 0.45, 0)
    a0.Parent = shard
    local a1 = Instance.new("Attachment")
    a1.Position = Vector3.new(0, shard.Size.Y * 0.45, 0)
    a1.Parent = shard
    local trail = Instance.new("Trail")
    trail.Name = "Capri_FragmentTrail"
    trail.Attachment0, trail.Attachment1 = a0, a1
    trail.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, PALETA.Verde),
        ColorSequenceKeypoint.new(0.65, PALETA.Magenta),
        ColorSequenceKeypoint.new(1, PALETA.Vermelho),
    })
    trail.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.12),
        NumberSequenceKeypoint.new(1, 1),
    })
    trail.WidthScale = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.7),
        NumberSequenceKeypoint.new(1, 0),

    trail.Lifetime = 0.22
    trail.MinLength = 0.04
    trail.FaceCamera = true
    trail.LightEmission = 1
    trail.Parent = shard

    return {
        part = shard,
        phase = ((indice - 1) / math.max(total, 1)) * math.pi * 2,
        radius = rf(1.05, 1.65),
        height = rf(0.45, 2.45),
        speed = rf(0.75, 1.35) * ((indice % 2 == 0) and -1 or 1),

capriIniciarAura = function(character)
    capriLimparAura()
    if not CONFIG.CapriAura.Ativo or encerrado or not character or not character.Parent then return end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    local torso = character:FindFirstChild("Torso")
    local head = character:FindFirstChild("Head")
    if not humanoid or humanoid.Health <= 0 or not root or not torso or humanoid.RigType ~= Enum.HumanoidRigType.R6 then return end

    local folder = Instance.new("Folder")
    folder.Name = "Capri_PersistentAura"
    folder.Parent = fxFolder
    CAPRI_AURA_STATE.character = character
    CAPRI_AURA_STATE.folder = folder

    if CONFIG.CapriAura.Silhueta then
        CAPRI_AURA_STATE.highlight = novo("Highlight", {
            Name = "Capri_EntitySilhouette",
            Adornee = character,
            FillColor = PALETA.Vermelho,
            OutlineColor = PALETA.Magenta,
            FillTransparency = 0.91,
            OutlineTransparency = 0.12,
            DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
        }, folder)
    end

    local core = novaParte({
        Name = "Capri_AuraCore",
        Size = Vector3.new(0.2, 0.2, 0.2),
        CFrame = root.CFrame,
        Transparency = 1,
        Material = Enum.Material.Plastic,
    })
    core.Parent = folder
    if CONFIG.CapriAura.Brilho then
        CAPRI_AURA_STATE.light = novo("PointLight", {
            Name = "Capri_LivingLight",
            Color = PALETA.Verde,
            Brightness = 2.8,
            Range = 15,
            Shadows = false,
        }, core)
    end

    if CONFIG.CapriAura.Particulas then
        CAPRI_AURA_STATE.attachments[#CAPRI_AURA_STATE.attachments + 1] = capriCriarParticulaAura(
            torso, { PALETA.Verde, PALETA.Magenta, PALETA.Ciano }, N(13 * CONFIG.CapriAura.Intensidade), 0.22, 1.8
        )
        if head then
            CAPRI_AURA_STATE.attachments[#CAPRI_AURA_STATE.attachments + 1] = capriCriarParticulaAura(
                head, { PALETA.Ciano, PALETA.Magenta, PALETA.Branco }, N(5 * CONFIG.CapriAura.Intensidade), 0.13, 1.1
            )
        end
    end

    local quantidade = math.clamp(N(CONFIG.CapriAura.Fragmentos), 3, 8)
    for i = 1, quantidade do
        CAPRI_AURA_STATE.shards[#CAPRI_AURA_STATE.shards + 1] = capriCriarFragmentoAura(folder, i, quantidade)
    end

    local iniciou = os.clock()
    CAPRI_AURA_STATE.connection = RunService.RenderStepped:Connect(function()
        if encerrado or not character.Parent or player.Character ~= character or humanoid.Health <= 0 or not root.Parent then
            capriLimparAura()
            return
        end
        local t = os.clock() - iniciou
        local centro = root.Position + Vector3.new(0, 1.15, 0)
        for i, item in ipairs(CAPRI_AURA_STATE.shards) do
            local part = item.part
            if part and part.Parent then
                local ang = t * item.speed + item.phase
                local raio = item.radius + math.sin(t * 1.7 + item.phase) * 0.18
                local pos = centro + Vector3.new(math.cos(ang) * raio, item.height + math.sin(t * 1.9 + item.phase) * 0.32, math.sin(ang) * raio)
                local look = CFrame.lookAt(pos, centro)
                part.CFrame = look * CFrame.Angles(math.sin(t * 2 + i) * 0.5, t * item.spin, math.cos(t * 1.4 + i) * 0.35)
            end
        end
        if CAPRI_AURA_STATE.light and CAPRI_AURA_STATE.light.Parent then
            CAPRI_AURA_STATE.light.Color = (math.sin(t * 1.3) > 0.15) and PALETA.Verde or PALETA.Magenta
            CAPRI_AURA_STATE.light.Brightness = 2.4 + (0.8 * (0.5 + 0.5 * math.sin(t * 2.2))) * CONFIG.CapriAura.Intensidade
            CAPRI_AURA_STATE.light.Range = 13 + 3 * (0.5 + 0.5 * math.sin(t * 1.5))
        end
        if CAPRI_AURA_STATE.highlight and CAPRI_AURA_STATE.highlight.Parent then
            CAPRI_AURA_STATE.highlight.FillTransparency = 0.90 + 0.035 * (0.5 + 0.5 * math.sin(t * 2.1))
            CAPRI_AURA_STATE.highlight.OutlineTransparency = 0.08 + 0.12 * (0.5 + 0.5 * math.sin(t * 3.1))
        end
        if core and core.Parent then core.CFrame = root.CFrame * CFrame.new(0, 1.1, 0) end
    end)
end

local function capriTweenDestruir(obj, duracao, propriedades, estilo)
    if not obj or not obj.Parent then return end
    tw(obj, duracao, propriedades, estilo or Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
    task.delay(duracao + 0.15, function()
        if obj and obj.Parent then pcall(function() obj:Destroy() end) end
    end)
end

-- 3) Estilhaços de luz verdes/rosas que atravessam o espaço ao redor do corpo.

    local quantidade = N(math.random(3, 6))

    for _ = 1, quantidade do
        local ang = rf(0, math.pi * 2)
        local direcao = Vector3.new(math.cos(ang), rf(-0.35, 0.8), math.sin(ang))
        if direcao.Magnitude < 0.01 then direcao = Vector3.new(1, 0, 0) end
        direcao = direcao.Unit

        local comprimento = rf(2.5, 6.5) * CONFIG.CapriAmbient.Intensidade
        local p0 = centro + direcao * rf(0.35, 1.0)
        local p1 = p0 + direcao * comprimento
        local meio = (p0 + p1) * 0.5
        local dur = rf(0.18, 0.42)

        local raio = novaParte({
            Name = "Capri_LightShard",
            Size = Vector3.new(rf(0.035, 0.09), rf(0.035, 0.09), comprimento),
            CFrame = CFrame.lookAt(meio, p1),
            Color = (math.random() < 0.72) and PALETA.Verde or PALETA.Magenta,
            Transparency = 0.08,
        })
        capriTweenDestruir(raio, dur, {
            Size = Vector3.new(0.01, 0.01, comprimento * 1.25),
            CFrame = raio.CFrame * CFrame.new(0, 0, -comprimento * 0.12),
            Transparency = 1,

    end

local CAPRI_EVENTOS = {
    capriSilhueta,
    capriFragmentos,
    capriRaios,
    capriOnda,
    capriPulsoLuz,
    capriGlitch,
}

local function capriManifestar()
    local character, root = capriObterPersonagem()
    if not character or not root then return end

    -- Escolhe efeitos sem repetição dentro da mesma manifestação.
    local pool = {}
    for i, fn in ipairs(CAPRI_EVENTOS) do pool[i] = fn end
    local total = math.clamp(
        math.random(CONFIG.CapriAmbient.EfeitosMin, CONFIG.CapriAmbient.EfeitosMax),
        1,
        #pool
    )

    for i = 1, total do
        local indice = math.random(1, #pool)
        local efeito = table.remove(pool, indice)
        task.delay(rf(0, 0.22), function()
            if encerrado then return end
            local charAtual, rootAtual = capriObterPersonagem()
            if charAtual and rootAtual then
                local ok, err = pcall(efeito, charAtual, rootAtual)
                if not ok and CONFIG.DEBUG then warn("[Capri FX]", err) end
            end
        end)
    end
end

        if not encerrado and CONFIG.CapriAmbient.Ativo then capriManifestar() end
    end
end)

local ultimoAtaqueVoz = 0
local ultimoImpactoVoz = 0

local function capriEfeitoAtaque()
    local character, root = capriObterPersonagem()
    if not character or not root then return end
    -- Burst curto para acompanhar a entrada do ataque sem lotar a tela.
    capriFragmentos(character, root)
    capriRaios(character, root)
    capriGlitch(character, root)
    camShake(0.22, 0.14, false)
end

capriOnImpact = function(character)
    local agora = os.clock()
    if agora - ultimoImpactoVoz < 1.8 or encerrado then return end
    ultimoImpactoVoz = agora
    tocarVoz("reaction", 0.9, rf(0.96, 1.04))
    local atual, root = capriObterPersonagem()
    if atual and root then
        capriSilhueta(atual, root)
        capriFragmentos(atual, root)
        flashTela(PALETA.Magenta, 0.09, 0.82)
    end
end

capriOnDodge = function(character)
    local agora = os.clock()
    if agora - ultimoImpactoVoz < 2.25 or encerrado then return end
    ultimoImpactoVoz = agora
    tocarVoz("reaction", 0.82, rf(0.98, 1.06))
    local atual, root = capriObterPersonagem()
    if atual and root then
        capriRaios(atual, root)
        aberracao(0.16, 4, 0.18)
    end
end

-- O script base não contém hitbox/combate próprio. Portanto, os taunts são
-- ligados ao input de ataque local (clique/toque não processado), não a uma
-- confirmação de acerto em outro jogador.
rastrear(UserInputService.InputBegan:Connect(function(input, processado)
    if encerrado or not CONFIG.Audio.Ativo or not CONFIG.Audio.FalasAtaque or not CONFIG.Audio.AtivarPorClique then return end
    local tipo = input.UserInputType
    local toque = (tipo == Enum.UserInputType.Touch)
    local clique = (tipo == Enum.UserInputType.MouseButton1)
    local gamepad = (tipo == Enum.UserInputType.Gamepad1 and input.KeyCode == Enum.KeyCode.ButtonR2)
    if not (clique or toque or gamepad) then return end
    if processado and not toque then return end
    -- espera 1 frame: se o toque/clique foi em cima do MENU, não é ataque
    task.defer(function()
        if encerrado then return end
        local tg = guiInputs[input]
        if tg and os.clock() - tg < 0.25 then return end
        local agora = os.clock()
        if agora - ultimoAtaqueVoz < 1.65 then return end
        ultimoAtaqueVoz = agora

        if math.random() < 0.48 then
            tocarVoz("preAttack", 0.92, rf(0.97, 1.04))
        else
            task.delay(0.32, function()
                if not encerrado then tocarVoz("taunt", 0.98, rf(0.96, 1.05)) end
            end)
        end
        capriEfeitoAtaque()
    end)
end))

-- API local opcional: permite chamar falas extras quando o conteúdo for confirmado.
genv.__CapriProtocol_PlayVoice = function(chave)
    if type(chave) ~= "string" then return false end
    if chave == "intro" or chave == "taunt" or chave == "reaction" or chave == "preAttack" then
        tocarVoz(chave)
        return true
    end
    if chave == "extra1" or chave == "extra2" then
        if not CONFIG.Audio.AudiosDesconhecidos then return false end
        tocarVoz(chave)
        return true
    end
    return false
end

genv.__CapriProtocol_Cleanup = function()
    encerrado = true
    salvandoAtivo = false
    emMorte = false
    for _, c in ipairs(conexoes) do pcall(function() c:Disconnect() end) end
    conexoes = {}
    limparConexoes()
    if hudConn then pcall(function() hudConn:Disconnect() end) end
    if MN.loopConn then pcall(function() MN.loopConn:Disconnect() end) end
    freezeAtivo = false
    freezeToken = freezeToken + 1
    limparFreeze()
    CamFX.shakes, CamFX.fov, CamFX.congelar, CamFX.custom, CamFX.ligado = {}, nil, nil, nil, false
    pcall(function() RunService:UnbindFromRenderStep(CAM_BIND) end)
    local camera = workspace.CurrentCamera
    if camera and fovOriginal then pcall(function() camera.FieldOfView = fovOriginal end) end
    pcall(capriLimparAura)
    if genv.__CapriProtocol_PlayVoice then genv.__CapriProtocol_PlayVoice = nil end
    for s in pairs(audiosAtivos) do pcall(function() s:Destroy() end) end
    for _, e in pairs(efl) do pcall(function() e:Destroy() end) end
    pcall(function() if marcador then marcador:Destroy() end end)
    pcall(function() fxFolder:Destroy() end)
    pcall(function() gui:Destroy() end)
end

if introPermitida then
    -- spawn animado: roda UMA vez por servidor (o morph e a aura entram dentro da intro)
    if charInicial then
        seguro(configurarCharacter, charInicial)
        seguro(iniciarIntroSegura, charInicial)
        local rp0 = charInicial:FindFirstChild("HumanoidRootPart")
        if rp0 then cframeSalvo = rp0.CFrame end
    else
        introEstado = "pendente"
    end
else
    -- sem intro (já rodou neste servidor, ou desligada): liga tudo direto
    if charInicial then
        seguro(configurarCharacter, charInicial)
        seguro(function()
            charInicial:WaitForChild("Humanoid", 10)
            task.wait(0.5)
            aplicarMorph()      -- 🎭 mantém o morph original configurado
        end)
        local rp0 = charInicial:FindFirstChild("HumanoidRootPart")
        if rp0 then cframeSalvo = rp0.CFrame end
    end
    if CONFIG.Menu.Ativo and CONFIG.Menu.AbrirAposIntro then
        task.delay(0.8, abrirMenu)
    end
    introEstado = "feita"
    task.delay(1.8, function()
        if encerrado then return end
        local personagem = player.Character
        if personagem and capriIniciarAura then capriIniciarAura(personagem) end
        tocarVoz("intro", 1.0, 1.0)
    end)
end
